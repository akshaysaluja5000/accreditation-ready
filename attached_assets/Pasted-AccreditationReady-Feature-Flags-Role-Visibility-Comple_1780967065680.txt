AccreditationReady — Feature Flags & Role Visibility
Complete Implementation Spec
For: Akshay | From: Rajit | Date: June 2026
OVERVIEW
This document covers two related builds:

Feature flags system — controls which platform features are visible per facility (education-only vs full platform, facilities role visibility)

Role visibility flags — controls which job roles appear on the role selection screen per facility

Both use the same database pattern — a JSONB features column on the facilities table. All flags default to the most restrictive setting (false) and must be explicitly enabled per facility.

PART 2 — BACKEND SERVICE
Create this file at services/features.service.js

// services/features.service.js

/**
 * Default feature flags — most restrictive setting
 * All features off unless explicitly enabled per facility
 */
const DEFAULT_FEATURES = {
  education:                    false,
  compliance:                   false,
  survey_readiness_agent:       false,
  content_intelligence_agent:   false,
  compliance_task_manager:      false,
  executive_readiness_agent:    false,
  wall_chart_tracker:           false,
  regulatory_watch_agent:       false,
};

/**
 * Default role visibility — all roles visible unless explicitly hidden
 * Exception: facilities_maintenance defaults to false
 */
const DEFAULT_ROLE_VISIBILITY = {
  or_circulating_nurse:          true,
  or_manager_charge_nurse:       true,
  scrub_tech_surgical_tech:      true,
  surgical_orthopedic_assistant: true,
  anesthesia_assistant_crna:     true,
  spd_technician:                true,
  pacu_floor_nurse:              true,
  environmental_services:        true,
  facilities_maintenance:        false,  // hidden by default — enterprise only
  compliance_officer_cno:        true,
  nurse_educator_staff_dev:      true,
};

/**
 * Get feature flags for a facility
 * Merges facility-specific flags with defaults
 */
async function getFacilityFeatures(db, facilityId) {
  const result = await db.query(
    'SELECT features, role_visibility FROM facilities WHERE id = $1',
    [facilityId]
  );

  if (!result.rows.length) {
    return {
      features:        DEFAULT_FEATURES,
      role_visibility: DEFAULT_ROLE_VISIBILITY,
    };
  }

  const { features, role_visibility } = result.rows[0];

  return {
    features: {
      ...DEFAULT_FEATURES,
      ...(features || {}),
    },
    role_visibility: {
      ...DEFAULT_ROLE_VISIBILITY,
      ...(role_visibility || {}),
    },
  };
}

/**
 * Check if a specific feature is enabled for a facility
 */
async function isFeatureEnabled(db, facilityId, featureKey) {
  const { features } = await getFacilityFeatures(db, facilityId);
  return features[featureKey] === true;
}

/**
 * Check if a specific role is visible for a facility
 */
async function isRoleVisible(db, facilityId, roleKey) {
  const { role_visibility } = await getFacilityFeatures(db, facilityId);
  return role_visibility[roleKey] === true;
}

/**
 * Get all visible roles for a facility
 * Returns array of role keys that are enabled
 */
async function getVisibleRoles(db, facilityId) {
  const { role_visibility } = await getFacilityFeatures(db, facilityId);
  return Object.entries(role_visibility)
    .filter(([, visible]) => visible === true)
    .map(([roleKey]) => roleKey);
}

module.exports = {
  getFacilityFeatures,
  isFeatureEnabled,
  isRoleVisible,
  getVisibleRoles,
  DEFAULT_FEATURES,
  DEFAULT_ROLE_VISIBILITY,
};
PART 3 — BACKEND ROUTES
Create this file at routes/features.routes.js

// routes/features.routes.js

const express = require('express');
const router = express.Router();
const featuresService = require('../services/features.service');
const { requireAuth, requireRole } = require('../middleware/auth');

/**
 * GET /api/features
 * Returns feature flags and visible roles for the current user's facility
 * Called on app load — frontend uses this to show/hide UI elements
 */
router.get('/', requireAuth, async (req, res) => {
  try {
    const { facilityId } = req.user;

    const { features, role_visibility } =
      await featuresService.getFacilityFeatures(req.db, facilityId);

    // Build list of visible roles for role selection screen
    const visibleRoles = Object.entries(role_visibility)
      .filter(([, visible]) => visible === true)
      .map(([roleKey]) => roleKey);

    res.json({
      features,
      role_visibility,
      visible_roles: visibleRoles,
    });
  } catch (err) {
    console.error('Error fetching features:', err);
    res.status(500).json({ error: 'Failed to fetch feature flags' });
  }
});

/**
 * GET /api/features/roles
 * Returns only the visible roles for this facility
 * Used by the role selection screen during onboarding
 */
router.get('/roles', requireAuth, async (req, res) => {
  try {
    const { facilityId } = req.user;
    const visibleRoles = await featuresService.getVisibleRoles(
      req.db,
      facilityId
    );
    res.json({ visible_roles: visibleRoles });
  } catch (err) {
    console.error('Error fetching visible roles:', err);
    res.status(500).json({ error: 'Failed to fetch roles' });
  }
});

/**
 * PATCH /api/features — Admin only
 * Update feature flags for a facility
 * Body: { features?: {}, role_visibility?: {} }
 */
router.patch('/',
  requireAuth,
  requireRole(['admin', 'super_admin']),
  async (req, res) => {
    try {
      const { facilityId } = req.user;
      const { features, role_visibility } = req.body;

      // Get current flags
      const current = await featuresService.getFacilityFeatures(
        req.db,
        facilityId
      );

      // Merge updates
      const updatedFeatures = features
        ? { ...current.features, ...features }
        : current.features;

      const updatedRoleVisibility = role_visibility
        ? { ...current.role_visibility, ...role_visibility }
        : current.role_visibility;

      await req.db.query(
        `UPDATE facilities
         SET features = $1, role_visibility = $2
         WHERE id = $3`,
        [
          JSON.stringify(updatedFeatures),
          JSON.stringify(updatedRoleVisibility),
          facilityId,
        ]
      );

      res.json({
        success:         true,
        features:        updatedFeatures,
        role_visibility: updatedRoleVisibility,
      });
    } catch (err) {
      console.error('Error updating features:', err);
      res.status(500).json({ error: 'Failed to update features' });
    }
  }
);

module.exports = router;

// utils/features.js
// Import this wherever feature flag checks are needed

/**
 * Fetch feature flags and visible roles for the current facility
 * Call once on app load and store in your global state/context
 */
export async function loadFacilityFeatures() {
  try {
    const response = await fetch('/api/features', {
      credentials: 'include',
    });
    const data = await response.json();
    return data;
    // Returns: { features, role_visibility, visible_roles }
  } catch (err) {
    console.error('Error loading features:', err);
    return null;
  }
}

/**
 * Check if a feature is enabled
 * Pass the features object from loadFacilityFeatures()
 */
export function isEnabled(features, featureKey) {
  return features?.[featureKey] === true;
}

/**
 * Check if a role should be shown on the role selection screen
 * Pass the role_visibility object from loadFacilityFeatures()
 */
export function isRoleVisible(roleVisibility, roleKey) {
  return roleVisibility?.[roleKey] === true;
}
