async function seed() {
  const client = await pool.connect();
  try {
    await client.query('BEGIN');

    // Step 1: confirm table structure
    // Adjust column names below if your schema uses different names
    const checkTable = await client.query(`
      SELECT column_name
      FROM information_schema.columns
      WHERE table_name = 'compliance_items'
    `);
    console.log('compliance_items columns:', checkTable.rows.map(r => r.column_name).join(', '));

    // Step 2: delete existing ASC items to avoid duplicates
    const deleted = await client.query(`
      DELETE FROM compliance_items WHERE module = 'ASC'
    `);
    console.log(`Deleted ${deleted.rowCount} existing ASC compliance items`);

    // Step 3: insert all items
    let inserted = 0;
    for (const item of items) {
      await client.query(`
        INSERT INTO compliance_items
          (code, name, tier, category, frequency, module, created_at, updated_at)
        VALUES
          ($1, $2, $3, $4, $5, $6, NOW(), NOW())
        ON CONFLICT (code, name) DO UPDATE SET
          tier = EXCLUDED.tier,
          category = EXCLUDED.category,
          frequency = EXCLUDED.frequency,
          module = EXCLUDED.module,
          updated_at = NOW()
      `, [item.code, item.name, item.tier, item.category, item.frequency, item.module]);
      inserted++;
    }

    await client.query('COMMIT');
    console.log(`\n✅ Successfully seeded ${inserted} ASC compliance items`);
    console.log('\nBreakdown by tier:');
    const tier1 = items.filter(i => i.tier === 'TIER_1').length;
    const tier2 = items.filter(i => i.tier === 'TIER_2').length;
    const tier3 = items.filter(i => i.tier === 'TIER_3').length;
    const tier4 = items.filter(i => i.tier === 'TIER_4').length;
    console.log(`  TIER_1 (recurring logs):     ${tier1}`);
    console.log(`  TIER_2 (document vault):     ${tier2}`);
    console.log(`  TIER_3 (wall postings):      ${tier3}`);
    console.log(`  TIER_4 (third-party certs):  ${tier4}`);
    console.log(`  TOTAL:                       ${items.length}`);

    // Step 4: verify
    const count = await client.query(`SELECT COUNT(*) FROM compliance_items WHERE module = 'ASC'`);
    console.log(`\nDatabase verification: ${count.rows[0].count} ASC items now in compliance_items table`);

  } catch (err) {
    await client.query('ROLLBACK');
    console.error('❌ Seed failed — rolled back:', err.message);
    console.error('\nIf column names are wrong, check the column list printed above');
    console.error('and update the INSERT statement to match your schema.');
  } finally {
    client.release();
    await pool.end();
  }
}
