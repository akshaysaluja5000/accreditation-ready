PART 5 — FRONTEND UTILITY
Create this file at utils/features.js

// utils/features.js
// Import this wherever feature flag checks are needed

/**
 * Fetch feature flags and visible roles for the current facility
 * Call once on app load and store in your global state/context
 */
export async function loadFacilityFeatures() {
  try {
    const response = await fetch('/api/features', {
      credentials: 'include',
    });
    const data = await response.json();
    return data;
    // Returns: { features, role_visibility, visible_roles }
  } catch (err) {
    console.error('Error loading features:', err);
    return null;
  }
}

/**
 * Check if a feature is enabled
 * Pass the features object from loadFacilityFeatures()
 */
export function isEnabled(features, featureKey) {
  return features?.[featureKey] === true;
}

/**
 * Check if a role should be shown on the role selection screen
 * Pass the role_visibility object from loadFacilityFeatures()
 */
export function isRoleVisible(roleVisibility, roleKey) {
  return roleVisibility?.[roleKey] === true;
}

6A. Load features on app startup
In your main App component or auth context, load features immediately after the user authenticates:

import { loadFacilityFeatures } from '../utils/features';

// In your auth/app initialization
useEffect(() => {
  if (user) {
    loadFacilityFeatures().then(featureData => {
      setFeatures(featureData.features);
      setRoleVisibility(featureData.role_visibility);
    });
  }
}, [user]);
